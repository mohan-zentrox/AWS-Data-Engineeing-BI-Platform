# Extra Python dependencies the sales_orders_pipeline DAG needs on top of a
# base apache-airflow image (see docker-compose.yml `airflow` service,
# which installs these plus libs/metadata_client and libs/dq_checks at
# container start for local dev).

# _load_warehouse_staging_table only uses create_engine/text/to_sql, which
# work the same on 1.4.
sqlalchemy>=1.4.36,<2.0
psycopg2-binary>=2.9,<3.0
